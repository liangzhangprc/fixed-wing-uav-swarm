cmake_minimum_required(VERSION 3.22)
project(fixed_wing_demo)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz_transport_vendor REQUIRED)
find_package(gz_msgs_vendor REQUIRED)
find_package(gz-sim REQUIRED)
find_package(gz-transport REQUIRED)
find_package(gz-msgs REQUIRED)
find_package(gz-plugin REQUIRED COMPONENTS register)
add_library(fixed_wing_dynamics SHARED src/fixed_wing_dynamics.cpp)
target_compile_features(fixed_wing_dynamics PRIVATE cxx_std_17)
target_link_libraries(fixed_wing_dynamics PRIVATE gz-sim::gz-sim gz-transport::gz-transport gz-msgs::gz-msgs gz-plugin::register)
install(TARGETS fixed_wing_dynamics LIBRARY DESTINATION lib)
install(PROGRAMS scripts/autopilot.py scripts/px4_node.py scripts/px4_mavlink_bridge.py scripts/swarm_coordinator.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/swarm_safety.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config worlds models third_party DESTINATION share/${PROJECT_NAME})
install(FILES README.md README_PX4_SITL.md README_SWARM.md LICENSE DESTINATION share/${PROJECT_NAME})
ament_package()
